class Inputs { }
class Output
{
    public string Path { get; set; }
}

static class Native
{
    [System.Runtime.InteropServices.DllImport("user32.dll")] static extern bool SetForegroundWindow(System.IntPtr h);
    [System.Runtime.InteropServices.DllImport("user32.dll")] static extern System.IntPtr SetActiveWindow(System.IntPtr h);
    public static void FocusRevit(System.IntPtr h) { SetForegroundWindow(h); SetActiveWindow(h); }
}

/// <summary>Opens a file picker dialog to select the architectural model file.</summary>
/// <param name="input">No inputs.</param>
/// <returns>The path chosen by the user.</returns>
Output Run(Inputs input)
{
    var dialog = new Microsoft.Win32.OpenFileDialog();
    dialog.Filter = "Revit Files (*.rvt)|*.rvt|All files (*.*)|*.*";
    Native.FocusRevit(UIApp.MainWindowHandle);
    string selectedPath = dialog.ShowDialog() == true ? dialog.FileName : "";
    return new Output { Path = selectedPath };
}
