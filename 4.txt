/// <summary>Reconstruit et normalise de manière idempotente (sans doublons) 100% des éléments structurels depuis la maquette Archi vers des familles STR.</summary>
/// <param name="input">Aucun paramètre requis.</param>
/// <returns>Statistiques des éléments reconstruits et normalisés.</returns>
Output Run(Inputs input)
{
    Document doc = UIDoc.Document;

    RevitLinkInstance linkInstance = new FilteredElementCollector(doc)
        .OfClass(typeof(RevitLinkInstance))
        .Cast<RevitLinkInstance>()
        .FirstOrDefault();

    if (linkInstance == null)
    {
        Logger.AppendLine("Erreur: Aucune maquette liée trouvée dans le projet hôte.");
        return new Output { Success = false, LinkTitle = "Aucun lien" };
    }

    Document linkDoc = linkInstance.GetLinkDocument();
    if (linkDoc == null)
    {
        Logger.AppendLine("Erreur: La maquette liée est déchargée.");
        return new Output { Success = false, LinkTitle = linkInstance.Name };
    }

    Transform totalTransform = linkInstance.GetTotalTransform();

    bool IsWoodElement(Element elem)
    {
        if (elem is Wall w)
        {
            string tName = w.WallType.Name;
            return tName.IndexOf("Bois", StringComparison.OrdinalIgnoreCase) >= 0 ||
                   tName.IndexOf("MOB", StringComparison.OrdinalIgnoreCase) >= 0 ||
                   tName.IndexOf("Ossature", StringComparison.OrdinalIgnoreCase) >= 0;
        }
        else if (elem is FamilyInstance fi)
        {
            string famName = fi.Symbol.Family.Name;
            string symName = fi.Symbol.Name;
            return famName.IndexOf("Bois", StringComparison.OrdinalIgnoreCase) >= 0 ||
                   symName.IndexOf("Bois", StringComparison.OrdinalIgnoreCase) >= 0;
        }
        return false;
    }

    bool HasStructuralLayer(Wall w)
    {
        if (w == null || w.WallType == null) return false;

        string wtName = w.WallType.Name.ToUpperInvariant();
        if (wtName.Contains("CLOISON") || wtName.Contains("DOUBLAGE") || wtName.Contains("FINITION") || wtName.Contains("PLACO"))
        {
            return false;
        }

        string[] porteurParamNames = new string[] { 
            "Porteur", "PORTEUR", "013EX_Porteur", "0013EX_Porteur", "0013EC_Porteur", "013EC_Porteur", "Portant", "PORTANT"
        };
        foreach (string pName in porteurParamNames)
        {
            Parameter p = w.LookupParameter(pName) ?? w.WallType.LookupParameter(pName);
            if (p != null && p.HasValue)
            {
                if (p.StorageType == StorageType.Integer && p.AsInteger() == 1) return true;
                if (p.StorageType == StorageType.Integer && p.AsInteger() == 0) return false;
                string sVal = p.AsString() ?? p.AsValueString() ?? "";
                if (sVal.Equals("OUI", StringComparison.OrdinalIgnoreCase) || sVal.Equals("TRUE", StringComparison.OrdinalIgnoreCase) || sVal.Equals("1")) return true;
                if (sVal.Equals("NON", StringComparison.OrdinalIgnoreCase) || sVal.Equals("FALSE", StringComparison.OrdinalIgnoreCase) || sVal.Equals("0")) return false;
            }
        }

        Parameter pStructUsage = w.get_Parameter(BuiltInParameter.WALL_STRUCTURAL_USAGE_PARAM);
        if (pStructUsage != null && pStructUsage.HasValue && pStructUsage.AsInteger() != 0) return true;

        Parameter pStructLookup = w.LookupParameter("Structure") ?? w.LookupParameter("Structurel");
        if (pStructLookup != null && pStructLookup.HasValue && pStructLookup.AsInteger() == 1) return true;

        CompoundStructure cs = w.WallType.GetCompoundStructure();
        if (cs != null)
        {
            IList<CompoundStructureLayer> layers = cs.GetLayers();
            foreach (CompoundStructureLayer layer in layers)
            {
                if (layer.Function == MaterialFunctionAssignment.Structure)
                {
                    return true;
                }
            }
        }

        if (wtName.Contains("STR_") || wtName.Contains("VOILE") || wtName.Contains("PORTEUR") || wtName.Contains("PORTANT") || wtName.Contains("BETON") || wtName.Contains("BÉTON") || wtName.Contains("C25") || wtName.Contains("C30"))
        {
            return true;
        }

        return false;
    }

    bool HasStructuralLayerFloor(Floor f)
    {
        if (f == null || f.FloorType == null) return false;

        string ftName = f.FloorType.Name.ToUpperInvariant();

        if (ftName.Contains("CARRELAGE") || ftName.Contains("PARQUET") || ftName.Contains("CHAPE") || 
            ftName.Contains("FINITION") || ftName.Contains("ISOLATION") || ftName.Contains("REVETEMENT") || 
            ftName.Contains("REVÊTEMENT") || ftName.Contains("FAUX PLANCHER") || ftName.Contains("MOQUETTE"))
        {
            return false;
        }

        string[] porteurParamNames = new string[] { 
            "Porteur", "PORTEUR", "013EX_Porteur", "0013EX_Porteur", "0013EC_Porteur", "013EC_Porteur"
        };
        foreach (string pName in porteurParamNames)
        {
            Parameter p = f.LookupParameter(pName) ?? f.FloorType.LookupParameter(pName);
            if (p != null && p.HasValue)
            {
                if (p.StorageType == StorageType.Integer && p.AsInteger() == 1) return true;
                if (p.StorageType == StorageType.Integer && p.AsInteger() == 0) return false;
                string sVal = p.AsString() ?? p.AsValueString() ?? "";
                if (sVal.Equals("OUI", StringComparison.OrdinalIgnoreCase) || sVal.Equals("TRUE", StringComparison.OrdinalIgnoreCase) || sVal.Equals("1")) return true;
                if (sVal.Equals("NON", StringComparison.OrdinalIgnoreCase) || sVal.Equals("FALSE", StringComparison.OrdinalIgnoreCase) || sVal.Equals("0")) return false;
            }
        }

        Parameter pStruct = f.get_Parameter(BuiltInParameter.FLOOR_PARAM_IS_STRUCTURAL);
        if (pStruct != null && pStruct.HasValue && pStruct.AsInteger() == 1) return true;

        CompoundStructure cs = f.FloorType.GetCompoundStructure();
        if (cs != null)
        {
            IList<CompoundStructureLayer> layers = cs.GetLayers();
            foreach (CompoundStructureLayer layer in layers)
            {
                if (layer.Function == MaterialFunctionAssignment.Structure)
                {
                    return true;
                }
            }
        }

        if (ftName.Contains("STR_") || ftName.Contains("DALLE") || ftName.Contains("CSP") || ftName.Contains("STRUCTURE") || ftName.Contains("BETON") || ftName.Contains("BÉTON") || ftName.Contains("RADIER") || ftName.Contains("C25") || ftName.Contains("C30"))
        {
            return true;
        }

        return false;
    }

    bool HasInnerLoops(Floor f)
    {
        if (f == null) return false;
        try
        {
            Options opts = new Options { DetailLevel = ViewDetailLevel.Fine };
            GeometryElement geom = f.get_Geometry(opts);
            if (geom != null)
            {
                foreach (GeometryObject obj in geom)
                {
                    if (obj is Solid solid && solid.Faces.Size > 0)
                    {
                        foreach (Face face in solid.Faces)
                        {
                            if (face is PlanarFace pf && Math.Abs(pf.FaceNormal.Z) > 0.9)
                            {
                                if (pf.EdgeLoops.Size > 1) return true;
                            }
                        }
                    }
                }
            }
        }
        catch { }
        return false;
    }

    void EnableStructuralParameter(Element elem)
    {
        if (elem == null) return;

        if (elem is Wall w)
        {
            Parameter pUsage = w.get_Parameter(BuiltInParameter.WALL_STRUCTURAL_USAGE_PARAM);
            if (pUsage != null && !pUsage.IsReadOnly)
            {
                try { if (pUsage.AsInteger() == 0) pUsage.Set(1); } catch { }
            }

            Parameter pStructLookup = w.LookupParameter("Structure") ?? w.LookupParameter("Structurel");
            if (pStructLookup != null && !pStructLookup.IsReadOnly)
            {
                try { pStructLookup.Set(1); } catch { }
            }
        }
        else if (elem is Floor f)
        {
            Parameter pStruct = f.get_Parameter(BuiltInParameter.FLOOR_PARAM_IS_STRUCTURAL);
            if (pStruct != null && !pStruct.IsReadOnly)
            {
                try { pStruct.Set(1); } catch { }
            }

            Parameter pStructLookup = f.LookupParameter("Structure") ?? f.LookupParameter("Structurel");
            if (pStructLookup != null && !pStructLookup.IsReadOnly)
            {
                try { pStructLookup.Set(1); } catch { }
            }
        }
        else
        {
            Parameter pStruct = elem.get_Parameter(BuiltInParameter.FLOOR_PARAM_IS_STRUCTURAL)
                             ?? elem.get_Parameter(BuiltInParameter.WALL_STRUCTURAL_USAGE_PARAM)
                             ?? elem.LookupParameter("Structure");
            if (pStruct != null && !pStruct.IsReadOnly)
            {
                try { pStruct.Set(1); } catch { }
            }
        }
    }

    string GetLocalisationValue(Element elem)
    {
        if (elem == null) return null;

        string ExtractParamValue(Parameter p)
        {
            if (p == null || !p.HasValue) return null;
            if (p.StorageType == StorageType.String)
            {
                string s = p.AsString();
                if (!string.IsNullOrWhiteSpace(s)) return s;
            }
            string vs = p.AsValueString();
            if (!string.IsNullOrWhiteSpace(vs)) return vs;
            if (p.StorageType == StorageType.Integer)
            {
                int valInt = p.AsInteger();
                return valInt.ToString();
            }
            if (p.StorageType == StorageType.Double)
            {
                return p.AsDouble().ToString();
            }
            return null;
        }

        string[] targetNames = new string[] { 
            "013EX_localisation", "0013EX_localisation", "013EX_Localisation", "0013EX_Localisation",
            "013EX_location", "0013EX_location", "013EX_Location", "0013EX_Location",
            "0013EC_Localisation", "013EC_Localisation", "13EC_Localisation", "Localisation", "LOCALISATION",
            "0013EC_Location", "013EC_Location", "13EC_Location", "Location", "LOCATION",
            "0013EC_Emplacement", "Emplacement", "EMPLACEMENT", "Repérage", "Reperage"
        };

        foreach (string tName in targetNames)
        {
            Parameter p = elem.LookupParameter(tName);
            string val = ExtractParamValue(p);
            if (!string.IsNullOrWhiteSpace(val)) return val;
        }

        foreach (Parameter p in elem.Parameters)
        {
            if (p == null || !p.HasValue) continue;
            string pName = p.Definition?.Name ?? "";
            if (pName.IndexOf("localisation", StringComparison.OrdinalIgnoreCase) >= 0 ||
                pName.IndexOf("location", StringComparison.OrdinalIgnoreCase) >= 0 ||
                pName.IndexOf("013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                pName.IndexOf("0013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                pName.IndexOf("013EC", StringComparison.OrdinalIgnoreCase) >= 0 ||
                pName.IndexOf("0013EC", StringComparison.OrdinalIgnoreCase) >= 0 ||
                pName.IndexOf("emplacement", StringComparison.OrdinalIgnoreCase) >= 0)
            {
                string val = ExtractParamValue(p);
                if (!string.IsNullOrWhiteSpace(val)) return val;
            }
        }

        ElementId typeId = elem.GetTypeId();
        if (typeId != ElementId.InvalidElementId)
        {
            Element typeElem = elem.Document.GetElement(typeId);
            if (typeElem != null)
            {
                foreach (string tName in targetNames)
                {
                    Parameter p = typeElem.LookupParameter(tName);
                    string val = ExtractParamValue(p);
                    if (!string.IsNullOrWhiteSpace(val)) return val;
                }

                foreach (Parameter p in typeElem.Parameters)
                {
                    if (p == null || !p.HasValue) continue;
                    string pName = p.Definition?.Name ?? "";
                    if (pName.IndexOf("localisation", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        pName.IndexOf("location", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        pName.IndexOf("013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        pName.IndexOf("0013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        pName.IndexOf("013EC", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        pName.IndexOf("0013EC", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        pName.IndexOf("emplacement", StringComparison.OrdinalIgnoreCase) >= 0)
                    {
                        string val = ExtractParamValue(p);
                        if (!string.IsNullOrWhiteSpace(val)) return val;
                    }
                }
            }
        }

        return null;
    }

    bool IsExteriorByLocalisationOrKeyword(Element elem)
    {
        if (elem == null) return false;

        int TestString(string str)
        {
            if (string.IsNullOrWhiteSpace(str)) return 0;
            string s = str.Trim().ToUpperInvariant();

            bool hasExt = s.Contains("EXTERIEUR") || s.Contains("EXTÉRIEUR") || s.Contains("EXTERIOR") || 
                         s.Contains("FACADE") || s.Contains("FAÇADE") || s.Contains("EXTERNE") || 
                         s.Contains("ENVELOPPE") || s.Contains("ENVÉLOPPE") || s.Contains("SOUBASSEMENT") ||
                         s == "EXT" || s.StartsWith("EXT_") || s.EndsWith("_EXT") || s.Contains("-EXT-") || 
                         s.Contains(" EXT ") || s.StartsWith("EXT ") || s.EndsWith(" EXT") || s.Contains(".EXT.") ||
                         s == "1";

            bool hasInt = s.Contains("INTERIEUR") || s.Contains("INTÉRIEUR") || s.Contains("INTERIOR") || 
                         s == "INT" || s.StartsWith("INT_") || s.EndsWith("_INT") || s.Contains("-INT-") || 
                         s.Contains(" INT ") || s.StartsWith("INT ") || s.EndsWith(" INT") || s.Contains(".INT.") ||
                         s == "0" || s == "2";

            if (hasExt && !hasInt) return 1;
            if (hasInt && !hasExt) return -1;
            if (hasExt && hasInt) return 1;

            return 0;
        }

        string locVal = GetLocalisationValue(elem);
        if (!string.IsNullOrWhiteSpace(locVal))
        {
            int rLoc = TestString(locVal);
            if (rLoc == 1) return true;
            if (rLoc == -1) return false;
        }

        ElementId typeId = elem.GetTypeId();
        if (typeId != ElementId.InvalidElementId)
        {
            Element typeElem = elem.Document.GetElement(typeId);
            if (typeElem != null)
            {
                int rType = TestString(typeElem.Name);
                if (rType == 1) return true;
                if (rType == -1) return false;

                if (typeElem is FamilySymbol sym && sym.Family != null)
                {
                    int rFam = TestString(sym.Family.Name);
                    if (rFam == 1) return true;
                    if (rFam == -1) return false;
                }

                if (typeElem is WallType wt)
                {
                    if (wt.Function == WallFunction.Exterior) return true;
                    if (wt.Function == WallFunction.Interior) return false;
                }
            }
        }

        foreach (Parameter p in elem.Parameters)
        {
            if (p == null || !p.HasValue) continue;
            string pVal = p.AsString() ?? p.AsValueString() ?? "";
            int rVal = TestString(pVal);
            if (rVal == 1) return true;
            if (rVal == -1) return false;
        }

        if (typeId != ElementId.InvalidElementId)
        {
            Element typeElem = elem.Document.GetElement(typeId);
            if (typeElem != null)
            {
                foreach (Parameter p in typeElem.Parameters)
                {
                    if (p == null || !p.HasValue) continue;
                    string pVal = p.AsString() ?? p.AsValueString() ?? "";
                    int rVal = TestString(pVal);
                    if (rVal == 1) return true;
                    if (rVal == -1) return false;
                }
            }
        }

        return false;
    }

    void CopyLocalisationValueSmart(Element sourceElem, Element targetElem)
    {
        if (sourceElem == null || targetElem == null) return;

        string val = GetLocalisationValue(sourceElem);
        if (string.IsNullOrEmpty(val)) return;

        string[] targetNames = new string[] { 
            "013EX_localisation", "0013EX_localisation", "013EX_Localisation", "0013EX_Localisation",
            "0013EC_Localisation", "013EC_Localisation", "13EC_Localisation", "Localisation",
            "013EX_location", "0013EX_location", "0013EC_Location", "013EC_Location", "Location"
        };

        Parameter pTarget = null;
        foreach (string tName in targetNames)
        {
            pTarget = targetElem.LookupParameter(tName);
            if (pTarget != null && !pTarget.IsReadOnly) break;
        }

        if (pTarget == null || pTarget.IsReadOnly)
        {
            foreach (Parameter p in targetElem.Parameters)
            {
                if (p != null && !p.IsReadOnly && p.Definition != null)
                {
                    string name = p.Definition.Name;
                    if (name.IndexOf("localisation", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        name.IndexOf("location", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        name.IndexOf("013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        name.IndexOf("0013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                        name.IndexOf("013EC", StringComparison.OrdinalIgnoreCase) >= 0)
                    {
                        pTarget = p;
                        break;
                    }
                }
            }
        }

        if (pTarget == null || pTarget.IsReadOnly)
        {
            ElementId typeId = targetElem.GetTypeId();
            if (typeId != ElementId.InvalidElementId)
            {
                Element typeElem = targetElem.Document.GetElement(typeId);
                if (typeElem != null)
                {
                    foreach (string tName in targetNames)
                    {
                        pTarget = typeElem.LookupParameter(tName);
                        if (pTarget != null && !pTarget.IsReadOnly) break;
                    }

                    if (pTarget == null || pTarget.IsReadOnly)
                    {
                        foreach (Parameter p in typeElem.Parameters)
                        {
                            if (p != null && !p.IsReadOnly && p.Definition != null)
                            {
                                string name = p.Definition.Name;
                                if (name.IndexOf("localisation", StringComparison.OrdinalIgnoreCase) >= 0 ||
                                    name.IndexOf("location", StringComparison.OrdinalIgnoreCase) >= 0 ||
                                    name.IndexOf("013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                                    name.IndexOf("0013EX", StringComparison.OrdinalIgnoreCase) >= 0 ||
                                    name.IndexOf("013EC", StringComparison.OrdinalIgnoreCase) >= 0)
                                {
                                    pTarget = p;
                                    break;
                                }
                            }
                        }
                    }
                }
            }
        }

        if (pTarget != null && !pTarget.IsReadOnly)
        {
            try
            {
                pTarget.Set(val);
            }
            catch { }
        }
    }

    XYZ GetElementLocationPoint(Element elem, Transform transform)
    {
        XYZ pt = null;
        if (elem.Location is LocationPoint locPt)
        {
            pt = transform.OfPoint(locPt.Point);
        }
        else if (elem.Location is LocationCurve locCurve)
        {
            pt = transform.OfPoint(locCurve.Curve.GetEndPoint(0));
        }
        else
        {
            BoundingBoxXYZ bb = elem.get_BoundingBox(null);
            if (bb != null)
            {
                pt = transform.OfPoint((bb.Min + bb.Max) * 0.5);
            }
        }
        return pt;
    }

        Level match = hostStructLevels.FirstOrDefault(l => string.Equals(l.Name.Trim(), targetName.Trim(), StringComparison.OrdinalIgnoreCase));

    }

    double GetThickestStructuralLayerWidth(Wall w)
    {
        if (w == null || w.WallType == null) return 0.2;
        CompoundStructure cs = w.WallType.GetCompoundStructure();
        if (cs != null)
        {
            IList<CompoundStructureLayer> layers = cs.GetLayers();
            double maxStructWidth = 0;

            foreach (CompoundStructureLayer layer in layers)
            {
                if (layer.Function == MaterialFunctionAssignment.Structure)
                {
                    if (layer.Width > maxStructWidth) maxStructWidth = layer.Width;
                }
            }
            if (maxStructWidth > 0.001) return maxStructWidth;

            int cStart = cs.GetFirstCoreLayerIndex();
            int cEnd = cs.GetLastCoreLayerIndex();
            if (cStart >= 0 && cEnd >= cStart && cEnd < layers.Count)
            {
                for (int i = cStart; i <= cEnd; i++)
                {
                    if (layers[i].Width > maxStructWidth) maxStructWidth = layers[i].Width;
                }
                if (maxStructWidth > 0.001) return maxStructWidth;
            }

            foreach (CompoundStructureLayer layer in layers)
            {
                if (layer.Width > maxStructWidth) maxStructWidth = layer.Width;
            }
            if (maxStructWidth > 0.001) return maxStructWidth;
        }
        return w.Width;
    }

    WallType GetOrCreateStrWallType(Document d, double widthFt, bool isExterior)
    {
        double widthCm = Math.Round(UnitUtils.ConvertFromInternalUnits(widthFt, UnitTypeId.Centimeters));
        if (widthCm <= 0) widthCm = 20;
        string suffix = isExterior ? "EXT" : "INT";

        WallType existing = strWallTypes.FirstOrDefault(wt => wt != null &&
            Math.Abs(UnitUtils.ConvertFromInternalUnits(wt.Width, UnitTypeId.Centimeters) - widthCm) < 0.2 &&
            wt.Name.EndsWith(suffix, StringComparison.OrdinalIgnoreCase));
        if (existing != null)
        {
            CompoundStructure csEx = existing.GetCompoundStructure();
            if (csEx != null && csEx.LayerCount > 1)
            {
                try
                {
                    CompoundStructure singleCs = CompoundStructure.CreateSingleLayerCompoundStructure(MaterialFunctionAssignment.Structure, widthFt, ElementId.InvalidElementId);
                    existing.SetCompoundStructure(singleCs);
                }
                catch { }
            }
            return existing;
        }

        string newName = $"STR_VO_{widthCm}_C25_{suffix}";
        WallType existingByName = new FilteredElementCollector(d)
            .OfClass(typeof(WallType))
            .Cast<WallType>()
            .FirstOrDefault(wt => wt != null && string.Equals(wt.Name.Trim(), newName.Trim(), StringComparison.OrdinalIgnoreCase));

        if (existingByName != null)
        {
            CompoundStructure csEx = existingByName.GetCompoundStructure();
            if (csEx != null && csEx.LayerCount > 1)
            {
                try
                {
                    CompoundStructure singleCs = CompoundStructure.CreateSingleLayerCompoundStructure(MaterialFunctionAssignment.Structure, widthFt, ElementId.InvalidElementId);
                    existingByName.SetCompoundStructure(singleCs);
                }
                catch { }
            }
            if (!strWallTypes.Contains(existingByName)) strWallTypes.Add(existingByName);
            return existingByName;
        }

        WallType templateType = strWallTypes.FirstOrDefault(wt => wt != null && wt.Name.EndsWith(suffix, StringComparison.OrdinalIgnoreCase))
                             ?? strWallTypes.FirstOrDefault(wt => wt != null)
                             ?? baseStrWallType
                             ?? new FilteredElementCollector(d).OfClass(typeof(WallType)).Cast<WallType>().FirstOrDefault(wt => wt.Kind == WallKind.Basic);

        if (templateType != null)
        {
            try
            {
                WallType duplicated = templateType.Duplicate(newName) as WallType;
                if (duplicated != null)
                {
                    try
                    {
                        CompoundStructure singleCs = CompoundStructure.CreateSingleLayerCompoundStructure(MaterialFunctionAssignment.Structure, widthFt, ElementId.InvalidElementId);
                        duplicated.SetCompoundStructure(singleCs);
                    }
                    catch
                    {
                        CompoundStructure cs = duplicated.GetCompoundStructure();
                        if (cs != null && cs.LayerCount > 0)
                        {
                            int coreIndex = cs.GetFirstCoreLayerIndex();
                            if (coreIndex < 0 || coreIndex >= cs.LayerCount) coreIndex = 0;
                            cs.SetLayerWidth(coreIndex, widthFt);
                            duplicated.SetCompoundStructure(cs);
                        }
                    }
                    strWallTypes.Add(duplicated);
                    return duplicated;
                }
            }
            catch { }
        }

        WallType fallbackWithSuffix = strWallTypes.FirstOrDefault(wt => wt != null && wt.Name.EndsWith(suffix, StringComparison.OrdinalIgnoreCase))

    {

        FloorType existingByName = new FilteredElementCollector(d)

            .Cast<FloorType>()
            .FirstOrDefault(ft => ft != null && string.Equals(ft.Name.Trim(), targetName.Trim(), StringComparison.OrdinalIgnoreCase));

        if (existingByName != null)
        {
            return existingByName;
        }

        FloorType templateType = new FilteredElementCollector(d)
            .OfClass(typeof(FloorType))
            .Cast<FloorType>()
            .FirstOrDefault(ft => ft.Name.StartsWith("STR_CSP_", StringComparison.OrdinalIgnoreCase))
            ?? new FilteredElementCollector(d)
                .OfClass(typeof(FloorType))
                .Cast<FloorType>()
                .FirstOrDefault(ft => ft.Name.StartsWith("STR_", StringComparison.OrdinalIgnoreCase))
            ?? new FilteredElementCollector(d)
                .OfClass(typeof(FloorType))
                .Cast<FloorType>()
                .FirstOrDefault();

        if (templateType != null)
        {
            try
            {
                FloorType duplicated = templateType.Duplicate(targetName) as FloorType;
                if (duplicated != null)
                {
                    CompoundStructure cs = duplicated.GetCompoundStructure();

            }
            catch { }
        }

            {
                if (!existingByName.IsActive) existingByName.Activate();

                    try { copiedSym.Name = targetName; } catch { }
                    if (!copiedSym.IsActive) copiedSym.Activate();
                    return copiedSym;
                }
            }
        }
        catch { }

        return null;

                        {
                            existsInHost = true;
                            break;
                        }
                    }
                }
            }

            if (!existsInHost)
            {
                try
                {
                    ICollection<ElementId> copiedIds = ElementTransformUtils.CopyElements(linkDoc, new List<ElementId> { shaft.Id }, doc, totalTransform, copyShaftOpts);
                    if (copiedIds != null && copiedIds.Count > 0)
                    {
                        shaftsRebuilt++;
                    }
                }
                catch { }
            }
        }

        List<Floor> linkFloors = new FilteredElementCollector(linkDoc)
            .OfClass(typeof(Floor))
            .Cast<Floor>()
            .Where(f => f.FloorType != null && HasStructuralLayerFloor(f))
            .ToList();

        existingHostFloors = new FilteredElementCollector(doc)
            .OfClass(typeof(Floor))
            .Cast<Floor>()
            .ToList();

        Dictionary<Floor, XYZ> hostFloorCentroids = new Dictionary<Floor, XYZ>();
        foreach (Floor hf in existingHostFloors)
        {
            CheckCancellationRequested();
            BoundingBoxXYZ bb = hf.get_BoundingBox(null);
            if (bb != null) hostFloorCentroids[hf] = (bb.Min + bb.Max) * 0.5;
        }

        int flrStep = 0;
        foreach (Floor linkFlr in linkFloors)
        {
            CheckCancellationRequested();
            flrStep++;
            ProgressUpdate($"Reconstruction Dalles, {flrStep} / {linkFloors.Count}", flrStep, linkFloors.Count);

            bool isExterior = IsExteriorByLocalisationOrKeyword(linkFlr);

            BoundingBoxXYZ linkBb = linkFlr.get_BoundingBox(null);
            XYZ linkCentroid = linkBb != null ? totalTransform.OfPoint((linkBb.Min + linkBb.Max) * 0.5) : null;

            Floor createdFloor = null;
            if (linkCentroid != null)
            {
                var kvp = hostFloorCentroids.FirstOrDefault(x => x.Value.DistanceTo(linkCentroid) < 3.0);
                if (kvp.Key != null)
                {
                    createdFloor = kvp.Key;
                }
            }

            if (createdFloor != null && HasInnerLoops(linkFlr) && !HasInnerLoops(createdFloor))
            {
                try
                {
                    doc.Delete(createdFloor.Id);
                    hostFloorCentroids.Remove(createdFloor);
                    createdFloor = null;
                }
                catch { }
            }

            if (createdFloor == null)
            {
                try
                {
                    CopyPasteOptions options = new CopyPasteOptions();
                    options.SetDuplicateTypeNamesHandler(new DuplicateTypeNamesHandler());
                    ICollection<ElementId> copiedIds = ElementTransformUtils.CopyElements(linkDoc, new List<ElementId> { linkFlr.Id }, doc, totalTransform, options);
                    if (copiedIds != null && copiedIds.Count > 0)
                    {
                        createdFloor = doc.GetElement(copiedIds.First()) as Floor;
                        if (createdFloor != null && linkCentroid != null)
                        {
                            hostFloorCentroids[createdFloor] = linkCentroid;
                        }
                    }
                }
                catch { }
            }

            if (createdFloor != null)
            {
                double thick = linkFlr.FloorType?.GetCompoundStructure()?.GetWidth()
                            ?? createdFloor.FloorType?.GetCompoundStructure()?.GetWidth()
                            ?? UnitUtils.ConvertToInternalUnits(20, UnitTypeId.Centimeters);

                FloorType targetType = GetOrCreateStrFloorType(doc, thick, isExterior);
                if (targetType != null && createdFloor.IsValidType(targetType.Id))
                {
                    try { createdFloor.ChangeTypeId(targetType.Id); } catch { }
                }

                if (linkLevelToHostStructLevel.TryGetValue(linkFlr.LevelId, out Level hostLevel))
                {
                    Parameter pLevel = createdFloor.get_Parameter(BuiltInParameter.LEVEL_PARAM);
                    if (pLevel != null && !pLevel.IsReadOnly)
                    {
                        try { pLevel.Set(hostLevel.Id); } catch { }
                    }
                }

                Parameter pOff = createdFloor.get_Parameter(BuiltInParameter.FLOOR_HEIGHTABOVELEVEL_PARAM);
                if (pOff != null && !pOff.IsReadOnly) pOff.Set(0.0);

                EnableStructuralParameter(createdFloor);

                CopyLocalisationValueSmart(linkFlr, createdFloor);
                floorsRebuilt++;
            }
        }

        List<Floor> remainingHostFloors = new FilteredElementCollector(doc)
            .OfClass(typeof(Floor))
            .Cast<Floor>()
            .Where(f => f.FloorType != null && !f.FloorType.Name.StartsWith("STR_", StringComparison.OrdinalIgnoreCase))
            .ToList();

        foreach (Floor hf in remainingHostFloors)
        {
            CheckCancellationRequested();
            double thick = hf.FloorType?.GetCompoundStructure()?.GetWidth() 
                        ?? UnitUtils.ConvertToInternalUnits(20, UnitTypeId.Centimeters);
            bool isExterior = IsExteriorByLocalisationOrKeyword(hf);
            FloorType targetType = GetOrCreateStrFloorType(doc, thick, isExterior);
            if (targetType != null && hf.IsValidType(targetType.Id))
            {
                try { hf.ChangeTypeId(targetType.Id); } catch { }
            }
            EnableStructuralParameter(hf);
        }

        foreach (Floor hf in existingHostFloors)
        {
            if (hf != null && HasStructuralLayerFloor(hf))
            {
                EnableStructuralParameter(hf);
            }
        }

        List<FamilyInstance> linkDoors = new FilteredElementCollector(linkDoc)
            .OfCategory(BuiltInCategory.OST_Doors)
            .OfClass(typeof(FamilyInstance))
            .Cast<FamilyInstance>()
            .ToList();

        int doorStep = 0;
        foreach (FamilyInstance linkDoor in linkDoors)
        {
            CheckCancellationRequested();
            doorStep++;
            ProgressUpdate($"Convert Portes STR, {doorStep} / {linkDoors.Count}", doorStep, linkDoors.Count);

            XYZ doorPt = GetElementLocationPoint(linkDoor, totalTransform);
            if (doorPt == null) continue;

            Wall hostWall = null;
            if (linkDoor.Host is Wall linkHostW && linkWallToHostWallMap.TryGetValue(linkHostW.Id, out var mappedW))
            {
                hostWall = mappedW;
            }

            if (hostWall == null && createdHostWalls.Count > 0)
            {
                double minDist = double.MaxValue;
                Wall bestW = null;
                foreach (Wall w in createdHostWalls)
                {
                    if (w.Location is LocationCurve lc)
                    {
                        double d = lc.Curve.Distance(doorPt);
                        if (d < minDist)
                        {
                            minDist = d;
                            bestW = w;
                        }
                    }
                }
                if (minDist <= 2.5) hostWall = bestW;
            }

            if (hostWall == null) continue;

            Level hostLevel = linkLevelToHostStructLevel.TryGetValue(linkDoor.LevelId, out var hL) ? hL : hostStructLevels.FirstOrDefault();
            if (hostLevel == null) continue;

            bool isExterior = IsExteriorByLocalisationOrKeyword(linkDoor);
            FamilySymbol doorSym = GetOrCreateStrDoorSymbol(doc, linkDoc, linkDoor, isExterior);
            if (doorSym == null) continue;

            XYZ placePt = doorPt;
            if (hostWall.Location is LocationCurve hostLc)
            {
                IntersectionResult proj = hostLc.Curve.Project(doorPt);
                if (proj != null) placePt = proj.XYZPoint;
            }

            FamilyInstance newDoor = existingHostDoors.FirstOrDefault(hd => {
                XYZ hPt = GetElementLocationPoint(hd, Transform.Identity);
                return hPt != null && hPt.DistanceTo(placePt) < 0.5;
            });

            if (newDoor == null)
            {
                try
                {
                    if (!doorSym.IsActive) doorSym.Activate();
                    newDoor = doc.Create.NewFamilyInstance(placePt, doorSym, hostWall, hostLevel, Autodesk.Revit.DB.Structure.StructuralType.NonStructural);
                    if (newDoor != null) existingHostDoors.Add(newDoor);
                }
                catch { }
            }
            else
            {
                if (doorSym != null)
                {
                    try
                    {
                        if (!doorSym.IsActive) doorSym.Activate();
                        newDoor.ChangeTypeId(doorSym.Id);
                    }
                    catch { }
                }
            }

            if (newDoor != null)
            {
                Parameter pSill = linkDoor.get_Parameter(BuiltInParameter.INSTANCE_SILL_HEIGHT_PARAM);
                if (pSill != null && pSill.HasValue)
                {
                    Parameter pNewSill = newDoor.get_Parameter(BuiltInParameter.INSTANCE_SILL_HEIGHT_PARAM);
                    if (pNewSill != null && !pNewSill.IsReadOnly)
                    {
                        try { pNewSill.Set(pSill.AsDouble()); } catch { }
                    }
                }

                CopyLocalisationValueSmart(linkDoor, newDoor);
                doorsRebuilt++;
            }
        }

        List<FamilyInstance> linkWinList = new FilteredElementCollector(linkDoc)
            .OfCategory(BuiltInCategory.OST_Windows)
            .OfClass(typeof(FamilyInstance))
            .Cast<FamilyInstance>()
            .ToList();

        int winStep = 0;
        foreach (FamilyInstance linkWin in linkWinList)
        {
            CheckCancellationRequested();
            winStep++;
            ProgressUpdate($"Convert Fenêtres STR, {winStep} / {linkWinList.Count}", winStep, linkWinList.Count);

            XYZ winPt = GetElementLocationPoint(linkWin, totalTransform);
            if (winPt == null) continue;

            Wall hostWall = null;
            if (linkWin.Host is Wall linkHostW && linkWallToHostWallMap.TryGetValue(linkHostW.Id, out var mappedW))
            {
                hostWall = mappedW;
            }

            if (hostWall == null && createdHostWalls.Count > 0)
            {
                double minDist = double.MaxValue;
                Wall bestW = null;
                foreach (Wall w in createdHostWalls)
                {
                    if (w.Location is LocationCurve lc)
                    {
                        double d = lc.Curve.Distance(winPt);
                        if (d < minDist)
                        {
                            minDist = d;
                            bestW = w;
                        }
                    }
                }
                if (minDist <= 2.5) hostWall = bestW;
            }

            if (hostWall == null) continue;

            Level hostLevel = linkLevelToHostStructLevel.TryGetValue(linkWin.LevelId, out var hL) ? hL : hostStructLevels.FirstOrDefault();
            if (hostLevel == null) continue;

            bool isExterior = IsExteriorByLocalisationOrKeyword(linkWin);
            FamilySymbol winSym = GetOrCreateStrWinSymbol(doc, linkDoc, linkWin, isExterior);
            if (winSym == null) continue;

            XYZ placePt = winPt;
            if (hostWall.Location is LocationCurve hostLc)
            {
                IntersectionResult proj = hostLc.Curve.Project(winPt);
                if (proj != null) placePt = proj.XYZPoint;
            }

            FamilyInstance newWin = existingHostWindows.FirstOrDefault(hw => {
                XYZ hPt = GetElementLocationPoint(hw, Transform.Identity);
                return hPt != null && hPt.DistanceTo(placePt) < 0.5;
            });

            if (newWin == null)
            {
                try
                {
                    if (!winSym.IsActive) winSym.Activate();
                    newWin = doc.Create.NewFamilyInstance(placePt, winSym, hostWall, hostLevel, Autodesk.Revit.DB.Structure.StructuralType.NonStructural);
                    if (newWin != null) existingHostWindows.Add(newWin);
                }
                catch { }
            }
            else
            {
                if (winSym != null)
                {
                    try
                    {
                        if (!winSym.IsActive) winSym.Activate();
                        newWin.ChangeTypeId(winSym.Id);
                    }
                    catch { }
                }
            }

            if (newWin != null)
            {
                Parameter pSill = linkWin.get_Parameter(BuiltInParameter.INSTANCE_SILL_HEIGHT_PARAM);
                if (pSill != null && pSill.HasValue)
                {
                    Parameter pNewSill = newWin.get_Parameter(BuiltInParameter.INSTANCE_SILL_HEIGHT_PARAM);
                    if (pNewSill != null && !pNewSill.IsReadOnly)
                    {
                        try { pNewSill.Set(pSill.AsDouble()); } catch { }
                    }
                }

                CopyLocalisationValueSmart(linkWin, newWin);
                windowsRebuilt++;
            }
        }

        List<FamilyInstance> allHostDoorsAndWindows = new FilteredElementCollector(doc)
            .WherePasses(new ElementMulticategoryFilter(new List<BuiltInCategory> { BuiltInCategory.OST_Doors, BuiltInCategory.OST_Windows }))
            .WhereElementIsNotElementType()
            .OfClass(typeof(FamilyInstance))
            .Cast<FamilyInstance>()
            .ToList();

        int dwIndex = 0;
        foreach (FamilyInstance inst in allHostDoorsAndWindows)
        {
            CheckCancellationRequested();
            dwIndex++;
            ProgressUpdate($"Synchronisation INT/EXT portes & fenêtres, {dwIndex} / {allHostDoorsAndWindows.Count}", dwIndex, allHostDoorsAndWindows.Count);

            Wall host = inst.Host as Wall;
            if (host == null) continue;

            string wallName = host.WallType.Name.ToUpper();
            string wallLoc = (host.LookupParameter("013EC_Localisation")?.AsString() ?? "").ToUpper();

            bool isInt = wallName.Contains("_INT") || wallLoc.Contains("INT") || wallLoc.Contains("INTERIEUR");
            bool isExt = wallName.Contains("_EXT") || wallLoc.Contains("EXT") || wallLoc.Contains("EXTERIEUR") || wallLoc.Contains("FACADE");

            string targetKind = isInt ? "INT" : "EXT";
            string targetLocalisation = isInt ? "INTERIEUR" : "EXTERIEUR";

            FamilySymbol curSymbol = inst.Symbol;
            string curSymName = curSymbol.Name;

            string targetSuffix = "_" + targetKind;
            string oldSuffix = isInt ? "_EXT" : "_INT";

            if (curSymName.EndsWith(oldSuffix))
            {
                string targetSymName = curSymName.Substring(0, curSymName.Length - oldSuffix.Length) + targetSuffix;

                FamilySymbol targetSymbol = curSymbol.Family.GetFamilySymbolIds()
                    .Select(id => doc.GetElement(id) as FamilySymbol)
                    .FirstOrDefault(fs => fs != null && fs.Name.Equals(targetSymName, StringComparison.OrdinalIgnoreCase));

                if (targetSymbol == null)
                {
                    targetSymbol = curSymbol.Duplicate(targetSymName) as FamilySymbol;
                    if (targetSymbol != null && !targetSymbol.IsActive) targetSymbol.Activate();
                }

                if (targetSymbol != null)
                {
                    try { inst.Symbol = targetSymbol; } catch { }
                }
            }
            else if (!curSymName.EndsWith(targetSuffix))
            {
                string targetSymName = curSymName + targetSuffix;

                FamilySymbol targetSymbol = curSymbol.Family.GetFamilySymbolIds()
                    .Select(id => doc.GetElement(id) as FamilySymbol)
                    .FirstOrDefault(fs => fs != null && fs.Name.Equals(targetSymName, StringComparison.OrdinalIgnoreCase));

                if (targetSymbol == null)
                {
                    targetSymbol = curSymbol.Duplicate(targetSymName) as FamilySymbol;
                    if (targetSymbol != null && !targetSymbol.IsActive) targetSymbol.Activate();
                }

                if (targetSymbol != null)
                {
                    try { inst.Symbol = targetSymbol; } catch { }
                }
            }

            Parameter pLoc = inst.LookupParameter("013EC_Localisation");
            if (pLoc != null && !pLoc.IsReadOnly && pLoc.AsString() != targetLocalisation)
            {
                try { pLoc.Set(targetLocalisation); } catch { }
            }

            Parameter pFunc = inst.LookupParameter("03_ECS_Detail_Fonction");
            if (pFunc != null && !pFunc.IsReadOnly && pFunc.AsString() != targetKind)
            {
                try { pFunc.Set(targetKind); } catch { }
            }

            Parameter pTypeHote = inst.LookupParameter("*303EC_TypeHote");
            if (pTypeHote != null && !pTypeHote.IsReadOnly && pTypeHote.AsString() != targetKind)
            {
                try { pTypeHote.Set(targetKind); } catch { }
            }
        }

        t.Commit();
    }

Logger.AppendLine($"Reconstruction terminée (Mode Idempotent) : {wallsRebuilt} voiles structurels, {columnsRebuilt} poteaux, {floorsRebuilt} dalles, {doorsRebuilt} portes, {windowsRebuilt} fenêtres.");

    return new Output
    {
        Success = true,
        WallsRebuilt = wallsRebuilt,
        ColumnsRebuilt = columnsRebuilt,
        FloorsRebuilt = floorsRebuilt,
        DoorsRebuilt = doorsRebuilt,
        WindowsRebuilt = windowsRebuilt,
        WoodWallsPreserved = woodWallsPreserved,
        WoodColsPreserved = woodColsPreserved,
        LinkTitle = linkDoc.Title
    };
}
